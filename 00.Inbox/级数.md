设
$$
\frac{a_{n} }{ a_{n+ 1}}= 1+ \frac{r }{n } + o\left(\frac{1 }{n^{\varepsilon } }\right) 
$$

$$
\frac{\frac{1 }{n } }{\frac{1 }{n+ 1 } }= 1+ \frac{1 }{n } 
$$

$$
\frac{1 }{n \left(\ln n\right)^{p} } 
$$

$$
a_{n}= \frac{1 }{n\left(\ln n\right)^{p} } 
$$

$$
\frac{a_{n} }{a_{n+ 1} }= \frac{n+ 1\left(\ln \left(n+ 1\right)\right)^{p} }{n\left(\ln n\right)^{p} }= \left(1+ \frac{1 }{n }\right)\left(\ln \left(1+ \frac{1 }{n }\right)\right)^{p} 
$$


$$
\int _{2}^{\infty}\frac{1 }{x }\frac{1 }{\left(\ln x\right)^{p} }\,d x=  \int _{\ln 2}^{\infty}\frac{1 }{t^{p} }\,d t
$$
当$p\ge 1$时发散, 当$p< 1$时收敛.
